# cspell:ignore openclouddesktop opencloudandroid opencloudios opencloudadmin
# cspell:ignore opencloudspaceadmin opencloudguest opencloudusers spn webfinger

# group setup

kanidm group create opencloud_admins
kanidm group add-members opencloud_admins cloud_admins
kanidm group create opencloud_users
kanidm group add-members opencloud_users cloud_users

# oidc client setup
#
# One client serves all four apps — web, desktop, Android and iOS. kanidm mints a
# per-client issuer URL, its own endpoints and its own token signing keys, while
# OpenCloud validates every token against a single OC_OIDC_ISSUER. A client per app
# therefore yields tokens the proxy rejects. The apps are steered onto this one
# client by OpenCloud's own webfinger service, which hands each platform a
# client_id in place of the vendor-fixed one it ships with.
#
# kanidm's book warns against sharing a client, but that is about sharing across
# distinct services. These four are front-ends of one resource server, which is the
# book's own "supplemental redirects within the same application" case.
#
# Public, not confidential: every OpenCloud client is a public client on PKCE and
# the chart carries no client-secret setting, so nothing is carried back into the
# SOPS secret.

kanidm system oauth2 create-public openclouddesktop "OpenCloud" https://cloud-eval.vonarx.online/

kanidm system oauth2 enable-pkce openclouddesktop

# PROXY_USER_OIDC_CLAIM is preferred_username, which kanidm fills from spn
# unless the client prefers the short name.
kanidm system oauth2 prefer-short-username openclouddesktop

# web
kanidm system oauth2 add-redirect-url openclouddesktop https://cloud-eval.vonarx.online/
kanidm system oauth2 add-redirect-url openclouddesktop https://cloud-eval.vonarx.online/oidc-callback.html
kanidm system oauth2 add-redirect-url openclouddesktop https://cloud-eval.vonarx.online/oidc-silent-redirect.html

# mobile, as opaque origins — kanidm matches a non-http(s) redirect on scheme and
# authority, which is what the apps build from their own fixed scheme and host.
kanidm system oauth2 add-redirect-url openclouddesktop oc://android.opencloud.eu
kanidm system oauth2 add-redirect-url openclouddesktop oc://ios.opencloud.eu

# desktop, which listens on an ephemeral port and redirects to http://127.0.0.1:<port>.
# kanidm allows this on public clients only, where PKCE is mandatory.
kanidm system oauth2 enable-localhost-redirects openclouddesktop

# The claim values are not free choice: PROXY_ROLE_ASSIGNMENT_OIDC_CLAIM is
# `roles`, and the proxy maps opencloudAdmin / opencloudSpaceAdmin /
# opencloudUser / opencloudGuest onto its own roles.
kanidm system oauth2 update-claim-map openclouddesktop roles opencloud_admins opencloudAdmin
kanidm system oauth2 update-claim-map openclouddesktop roles opencloud_users opencloudUser

# The union of what the four platforms request; each asks for a subset, so only the
# sync clients spend offline_access. `roles` has to be granted as a scope as well as
# mapped as a claim — kanidm refuses a scope it has not mapped. `groups_name` is the
# scope that emits the `groups` claim with names; there is no `groups_names`.
kanidm system oauth2 update-scope-map openclouddesktop opencloud_admins openid profile email groups_name roles offline_access
kanidm system oauth2 update-scope-map openclouddesktop opencloud_users openid profile email groups_name roles offline_access
